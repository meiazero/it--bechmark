cmake_minimum_required(VERSION 3.20)
project(ofdm LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ITPP REQUIRED IMPORTED_TARGET itpp)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
add_executable(ofdm src/main.cpp src/dsp.cpp)
target_compile_features(ofdm PRIVATE cxx_std_20)
target_compile_options(ofdm PRIVATE -O3 -march=native -Wall -Wextra -Wpedantic)
target_link_libraries(ofdm PRIVATE PkgConfig::ITPP nlohmann_json::nlohmann_json Threads::Threads)
target_compile_definitions(ofdm PRIVATE PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(seed64_check tests/seed64_check.cpp src/dsp.cpp)
target_include_directories(seed64_check PRIVATE src)
target_compile_features(seed64_check PRIVATE cxx_std_20)
target_compile_options(seed64_check PRIVATE -O3 -march=native -Wall -Wextra -Wpedantic)
target_link_libraries(seed64_check PRIVATE PkgConfig::ITPP nlohmann_json::nlohmann_json)
